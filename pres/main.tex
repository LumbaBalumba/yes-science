\section{Введение и постановка задачи}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Основные понятия}
	\begin{itemize}
		\item \textbf{Биодоступность} - скорость и степень, с которой действующее вещество или активнеая часть молекулы абсорбирующего вещества абсорбируется из лекарственного препарата и становится доступной в месте своего действия.
		\item \textbf{Биоэквивалентность} - схожесть биодоступности между исследуемыми препаратами.
		\item \textbf{Фармакокинетика} - раздел фармакологии, изучающий воздействие организма на введённый лекарственный препарат через механизмы всасывания, распределения, метаболической трансформации и выведения.
	\end{itemize}
\end{frame}

\begin{frame}{Исследуемое явление}
	Исследуется проверка биоэквивалентности двух препаратов на некоторой группе. Каждому члену группы вводится препарат. Затем в течение некотрого времени (обычно несколько суток) проводится измерение количества вещества в крови пациента. В результате получается кривая <<концентрация - время>>.
\end{frame}

\section{Описание данных и моделей}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{PBFTPK}
	Физиологически обоснованная конечновременная фармакокинетическая модель (\textbf{PBFTPK}) \newline
	$\textbf{PBFTPK}_0$:
	\[
		C(t) =
		\begin{cases}
			\frac{FD}{\tau} \frac{1}{V_d k_{el}} (1 - e^{-k_{el} t}), & t \leq \tau \\
			C(\tau)e^{-k_{el}(t-\tau)},                               & t > \tau
		\end{cases}
	\]
	\vspace{0.5em}
	$\textbf{PBFTPK}_1$:
	\[
		C(t) =
		\begin{cases}
			\frac{FD k_a}{V_d (k_a - k_{el})} \Big(e^{-k_{el}t} - e^{-k_a t}\Big), & t \leq \tau \\
			C(\tau)e^{-k_{el}(t-\tau)},                                            & t > \tau
		\end{cases}
	\]
	\begin{itemize}
		\item Параметры: \( F, D, V_d, k_a, k_{el}, \tau \)
	\end{itemize}
\end{frame}

\begin{frame}{Описание данных}
	\begin{itemize}
		\item Данные представлены в виде временных рядов концентрации лекарства.
		\item Особенности: неравномерность замеров, разные порядки измерений, фазовый рост и спад концентрации.
		\item Зачастую отсутствие одного явного пика (в отличие от классических \textbf(PBFTPK)-моделей).
	\end{itemize}
	\begin{center}
		\includegraphics[width=0.55\linewidth]{sample.png}
	\end{center}
\end{frame}

\begin{frame}{Ансамблирование моделей}
	\textbf{EPBFTPK:}
	\begin{itemize}
		\item Ансамблирование моделей: построение \(\hat{C}_i(t)\) на основе остатков \(r_i(t) = C(t) - \sum_i \hat{C}_i(t)\).
		\item Итоговая оценка:
		      \[
			      \hat{C}(t) = \sum_i \hat{C}_i(t)
		      \]
	\end{itemize}
	\begin{itemize}
		\item Позволяет представить траекторию ввиде суммы конечного числа неависимых траекторий, что решает проблему множества пиков.
	\end{itemize}
\end{frame}

\begin{frame}{Кусочная модификация PBFTPK}
	\begin{itemize}
		\item Введение дополнительных параметров: \( \tau_0 \) и \( \tau_{max} \).
		\item Новая модель (на примере $\textbf{PBFTPK}_1$):
		      \[
			      C(t) = \begin{cases}
				      0,                                                                                   & 0 \leq t \le \tau_0 \\
				      \frac{FD k_a}{V_d (k_a - k_{el})} (e^{-k_{el}(t - \tau_0)} - e^{-k_a (t - \tau_0)}), & \tau_0 < t \le \tau \\
				      C(\tau)e^{-k_{el}((t - \tau_0)-\tau)},                                               & t > \tau
			      \end{cases}
		      \]
		\item Методы определения \( \tau \) и \( \tau_0 \):
		      \begin{itemize}
			      \item \textbf{Минимаксный метод:}
			            \[
				            \tau = \argmax_{t < \tau_{max}} C(t), \quad \tau_0 = \argmin_{t < \tau} C(t)
			            \]
			      \item \textbf{Пиковый метод:}
			            \[
				            \tau = \max\,\Argmax_{t < \tau_{max}} C(t), \quad \tau_0 = \max\,\Argmin_{t < \tau} C(t)
			            \]
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Функция потерь}
	\textbf{Функция потерь:} \(\lambda\)-взвешенная MSE
	\[
		L_{\lambda} = \frac{1}{N}\sum_{i=1}^{N} \Big(C(t_i)-\hat{C}(t_i)\Big)^2 \cdot
		\begin{cases}
			1,       & t_i \le \tau \\
			\lambda, & t_i > \tau
		\end{cases}
	\]
	\begin{itemize}
		\item Регулировка влияния ошибок на фазах абсорбции и элиминации.
		\item Позволяет решить проблему неоднородности данных по времени.
	\end{itemize}
\end{frame}

		L_{\lambda}(r_i, \hat{C}_i, t) \to \inf_{p \in \Omega}, \;
